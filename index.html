LÓGICA DE LOCALSTORAGE E TRAVA let isLocked = false; function carregarRascunhoOffline() { const dadosSalvos = localStorage.getItem('pcp_rascunho_offline'); if (dadosSalvos) { try { const dadosParseados = JSON.parse(dadosSalvos); const linhas = document.querySelectorAll('#drillingTableBody tr'); dadosParseados.forEach((dado, index) => { if (linhas[index]) { const inputsTempo = linhas[index].querySelectorAll('input[type="time"]'); if (inputsTempo.length >= 2) { inputsTempo[0].value = dado.hora_inicio; inputsTempo[1].value = dado.hora_fim; if (inputsTempo[2]) inputsTempo[2].value = dado.intervalo; } } }); } catch (e) { console.error("Erro ao ler rascunho offline"); } } } function salvarRascunhoOffline() { if (isLocked) return; const linhas = document.querySelectorAll('#drillingTableBody tr'); let dadosRascunho = []; linhas.forEach((linha, index) => { const inputsTempo = linha.querySelectorAll('input[type="time"]'); if (inputsTempo.length >= 2) { dadosRascunho.push({ dia: index + 1, hora_inicio: inputsTempo[0].value, hora_fim: inputsTempo[1].value, intervalo: inputsTempo[2] ? inputsTempo[2].value : '00:00' }); } }); localStorage.setItem('pcp_rascunho_offline', JSON.stringify(dadosRascunho)); } document.addEventListener('DOMContentLoaded', () => { carregarRascunhoOffline(); const todosInputsTempo = document.querySelectorAll('#drillingTableBody input[type="time"]'); todosInputsTempo.forEach(input => { input.addEventListener('change', salvarRascunhoOffline); }); }); async function toggleLockPCP() { const badge = document.getElementById('badge-status-pcp'); const btn = document.getElementById('btn-lock-pcp'); const btnText = document.getElementById('lock-btn-text'); const inputs = document.querySelectorAll('.time-input'); if (isLocked) { // Destravar btnText.textContent = "Destravando..."; if (dbCliente) { const { error } = await dbCliente.from('pcp_matriz_mensal') .update({ travado: false }) .eq('mes_referencia', '2026-01') .eq('setor', 'Perfuracao'); if (error) { alert("Erro ao destravar na nuvem: " + error.message); btnText.textContent = "Destravar Rascunho"; return; } } isLocked = false; badge.className = "inline-flex items-center gap-1.5 px-3 py-1 rounded-full text-xs font-bold bg-amber-100 text-amber-900 border border-amber-300"; badge.innerHTML = ' Em Rascunho'; btn.className = "inline-flex items-center gap-2 px-4 py-2 bg-blue-600 text-white rounded text-xs font-semibold shadow hover:bg-blue-700 transition-all"; btnText.textContent = "Travar PCP Original"; inputs.forEach(i => { i.disabled = false; i.classList.remove('opacity-70', 'cursor-not-allowed', 'bg-slate-100'); }); salvarRascunhoOffline(); } else { // Travar btnText.textContent = "Salvando no Banco..."; btn.classList.add('opacity-50', 'cursor-not-allowed'); const linhas = document.querySelectorAll('#drillingTableBody tr'); let planData = []; linhas.forEach((linha, index) => { const inputsTempo = linha.querySelectorAll('input[type="time"]'); if (inputsTempo.length >= 2) { planData.push({ dia: index + 1, hora_inicio: inputsTempo[0].value, hora_fim: inputsTempo[1].value, intervalo: inputsTempo[2] ? inputsTempo[2].value : '00:00' }); } }); if (dbCliente) { const { error } = await dbCliente .from('pcp_matriz_mensal') .upsert({ mes_referencia: '2026-01', setor: 'Perfuracao', dados_diarios: planData, travado: true, travado_por: 'Eng. Davi Apolinário', data_trava: new Date().toISOString() }, { onConflict: 'mes_referencia, setor' }); if (error) { alert("Erro ao salvar na nuvem. Os dados estão locais. Erro: " + error.message); btnText.textContent = "Travar PCP Original"; btn.classList.remove('opacity-50', 'cursor-not-allowed'); return; } } else { alert("Atenção: Supabase não configurado. O PCP foi travado visualmente, mas não salvo na nuvem."); } isLocked = true; localStorage.removeItem('pcp_rascunho_offline'); badge.className = "inline-flex items-center gap-1.5 px-3 py-1 rounded-full text-xs font-bold bg-emerald-100 text-emerald-900 border border-emerald-300"; badge.innerHTML = ' PCP Travado (Original)'; btn.className = "inline-flex items-center gap-2 px-4 py-2 bg-slate-700 text-white rounded text-xs font-semibold shadow hover:bg-slate-800 transition-all"; btnText.textContent = "Destravar Rascunho"; btn.classList.remove('opacity-50', 'cursor-not-allowed'); inputs.forEach(i => { i.disabled = true; i.classList.add('opacity-70', 'cursor-not-allowed', 'bg-slate-100'); }); } }
